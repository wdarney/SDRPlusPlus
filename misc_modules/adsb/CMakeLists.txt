cmake_minimum_required(VERSION 3.13)
project(adsb LANGUAGES C CXX)
if(NOT APPLE)
    message(FATAL_ERROR "The ADS-B tar1090 window currently supports macOS only")
endif()
enable_language(OBJCXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ADSB_RTL REQUIRED IMPORTED_TARGET librtlsdr)
set(DECODER_SOURCES src/decoder.c)
foreach(name demod_2400 mode_s mode_ac track cpr crc icao_filter comm_b ais_charset util)
    list(APPEND DECODER_SOURCES vendor/dump1090/${name}.c)
endforeach()
add_library(adsb_decoder STATIC ${DECODER_SOURCES})
set_target_properties(adsb_decoder PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(adsb_decoder PRIVATE vendor/dump1090 PUBLIC src)
target_compile_features(adsb_decoder PRIVATE c_std_11)

set(SRC src/main.cpp src/receiver.cpp src/http_server.cpp src/map_window.mm)
include(${SDRPP_MODULE_CMAKE})
target_compile_features(adsb PRIVATE cxx_std_17)
target_include_directories(adsb PRIVATE ${SDRPP_CORE_ROOT})
target_link_libraries(adsb PRIVATE adsb_decoder PkgConfig::ADSB_RTL "-framework Cocoa" "-framework WebKit")
set_source_files_properties(src/map_window.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
add_custom_target(adsb_web COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/web ${CMAKE_CURRENT_BINARY_DIR}/adsb-web)
add_dependencies(adsb adsb_web)
install(DIRECTORY web/ DESTINATION share/sdrpp/res/adsb)
option(OPT_ADSB_TESTS "Build ADS-B regression tests" OFF)
if(OPT_ADSB_TESTS)
    enable_testing()
    add_executable(adsb_history_test tests/history_test.cpp)
    target_include_directories(adsb_history_test PRIVATE src)
    target_compile_features(adsb_history_test PRIVATE cxx_std_17)
    target_compile_options(adsb_history_test PRIVATE -UNDEBUG)
    add_test(NAME adsb_history COMMAND adsb_history_test)
    add_executable(adsb_decoder_test tests/decoder_test.cpp)
    target_link_libraries(adsb_decoder_test PRIVATE adsb_decoder)
    add_test(NAME adsb_decoder COMMAND adsb_decoder_test)
    add_executable(adsb_http_fixture tests/http_fixture.cpp src/http_server.cpp src/receiver.cpp)
    target_include_directories(adsb_http_fixture PRIVATE ${SDRPP_CORE_ROOT} src)
    target_compile_features(adsb_http_fixture PRIVATE cxx_std_17)
    target_link_libraries(adsb_http_fixture PRIVATE adsb_decoder PkgConfig::ADSB_RTL)
    add_test(NAME adsb_receiver_lifecycle COMMAND adsb_http_fixture ${CMAKE_CURRENT_SOURCE_DIR}/web --receiver-check)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME adsb_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py
        $<TARGET_FILE:adsb_http_fixture> ${CMAKE_CURRENT_SOURCE_DIR}/web)
endif()
