cmake_minimum_required(VERSION 3.13)
project(vdl2_protocol_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include(GNUInstallDirs)
option(VDL2_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(VDL2_THREAD_SANITIZER "Enable ThreadSanitizer (separate from ASan)" OFF)
if(VDL2_THREAD_SANITIZER AND VDL2_SANITIZERS)
    message(FATAL_ERROR "Use separate ASan and TSan build directories")
endif()
if(VDL2_THREAD_SANITIZER)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()
if(VDL2_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
# Build the vendored library directly with system optional dependencies. This
# bypasses the unrelated, incomplete zlib packaging on the historical branch.
set(LA_VERSION "2.2.1")
set(LA_VERSION_MAJOR 2)
set(LA_VERSION_MINOR 2)
set(LA_VERSION_PATCH 1)
set(LA_SUBDIR "libacars-2")
include_directories("${REPO}/core/libacars")
add_subdirectory("${REPO}/core/libacars/libacars" libacars)
add_subdirectory("${REPO}/core/libcorrect" libcorrect)
add_subdirectory(../protocol protocol)
add_executable(protocol_tests protocol_tests.cpp ../src/vdl2_protocol.cpp ../src/vdl2_dsp.cpp ../src/vdl2_message_json.cpp ../src/aviation_jsonl.cpp ../src/acars_dsp.cpp ../src/adsb_dsp.cpp)
target_include_directories(protocol_tests PRIVATE ../src ../protocol
    "${CMAKE_CURRENT_BINARY_DIR}/protocol" "${REPO}/core/libacars/libacars/asn1" "${REPO}/core/src" "${REPO}/core/libcorrect/include")
target_link_libraries(protocol_tests PRIVATE vdl2_protocol correct_static)
enable_testing()
add_test(NAME vdl2_upper_layers COMMAND protocol_tests)

# Tests the same Handler ownership and phased stop helper as the module, using
# live producer/consumer threads and SDR++ streams (no radio/GUI required).
find_package(PkgConfig REQUIRED)
pkg_check_modules(VOLK REQUIRED IMPORTED_TARGET volk)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/main.cpp" MODULE_SOURCE)
string(REGEX MATCHALL "MODE_(VDL2|ACARS|ADSB),[ \t]+(true|false)" CHANNEL_ENTRIES "${MODULE_SOURCE}")
list(LENGTH CHANNEL_ENTRIES CONFIGURED_CHANNEL_COUNT)
add_executable(lifecycle_tests lifecycle_tests.cpp)
target_include_directories(lifecycle_tests PRIVATE ../src "${REPO}/core/src")
target_compile_definitions(lifecycle_tests PRIVATE VDL2_CONFIGURED_CHANNELS=${CONFIGURED_CHANNEL_COUNT})
target_link_libraries(lifecycle_tests PRIVATE PkgConfig::VOLK)
add_test(NAME vdl2_lifecycle COMMAND lifecycle_tests)
set_tests_properties(vdl2_lifecycle PROPERTIES TIMEOUT 120)
